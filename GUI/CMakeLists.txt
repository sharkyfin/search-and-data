cmake_minimum_required(VERSION 3.24)
project(indexing_ui LANGUAGES C CXX)

set(CMAKE_CXX_STANDARD 17)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
set(CMAKE_POLICY_VERSION_MINIMUM 3.5)
set(CMAKE_CXX_EXTENSIONS OFF)
get_filename_component(INDEXING_PROJECT "${CMAKE_CURRENT_SOURCE_DIR}/.." ABSOLUTE)

include(FetchContent)
set(BUILD_EXAMPLES OFF CACHE BOOL "" FORCE)
FetchContent_Declare(raylib URL https://github.com/raysan5/raylib/archive/refs/tags/5.5.tar.gz)
FetchContent_Declare(raygui URL https://github.com/raysan5/raygui/archive/refs/tags/4.0.tar.gz)
FetchContent_MakeAvailable(raylib raygui)

add_executable(indexing_ui main.cpp Application.cpp "${INDEXING_PROJECT}/benchmark/Benchmark.cpp")
target_include_directories(indexing_ui PRIVATE "${INDEXING_PROJECT}")
target_include_directories(indexing_ui SYSTEM PRIVATE ${raygui_SOURCE_DIR}/src)
if(MSVC)
    target_compile_options(indexing_ui PRIVATE /W4)
else()
    target_compile_options(indexing_ui PRIVATE -Wall -Wextra -Wpedantic)
endif()
target_link_libraries(indexing_ui PRIVATE raylib)
